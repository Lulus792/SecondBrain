function(sb_record_link_map target)
    if(NOT SB_RECORD_LINK_MAPS)
        return()
    endif()
    set(map "${SB_LINK_MAP_DIR}/$<CONFIG>/${target}.map")
    if(MSVC)
        target_link_options(${target} PRIVATE "/MAP:${map}" "/VERBOSE:LIB")
    elseif(APPLE)
        target_link_options(${target} PRIVATE "LINKER:-map,${map}")
    elseif(CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang)$")
        target_link_options(${target} PRIVATE "LINKER:-Map,${map}")
    else()
        message(FATAL_ERROR "Link-map auditing is not configured for this toolchain")
    endif()
    add_custom_command(TARGET ${target} PRE_LINK
        COMMAND "${CMAKE_COMMAND}" -E rm -f "${map}" "${map}.identity"
        VERBATIM)
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND "${CMAKE_COMMAND}" "-DSB_PROGRAM=$<TARGET_FILE:${target}>" "-DSB_MAP=${map}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tools/record_link_map.cmake"
        VERBATIM)
endfunction()

    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --show-toplevel
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE SB_GIT_ROOT
        OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --short=12 HEAD
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE SB_GIT_REVISION
        OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE SB_GIT_RESULT)
    if(SB_GIT_RESULT EQUAL 0 AND SB_GIT_ROOT STREQUAL CMAKE_CURRENT_SOURCE_DIR AND SB_GIT_REVISION MATCHES "^[0-9a-f]+$")
        set(SB_SOURCE_REVISION "${SB_GIT_REVISION}")
        execute_process(COMMAND "${GIT_EXECUTABLE}" diff --quiet HEAD --
            WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" RESULT_VARIABLE SB_GIT_DIRTY ERROR_QUIET)
        if(NOT SB_GIT_DIRTY EQUAL 0)
            string(APPEND SB_SOURCE_REVISION "-dirty")
        endif()
    endif()
endif()
