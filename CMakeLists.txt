cmake_minimum_required(VERSION 3.20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(SecondBrain VERSION 0.3.1 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SB_BUILD_UI "Build SDL desktop UI" ON)
include(CTest)

file(GLOB_RECURSE SB_TEMPLATE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/templates/brain/*.md")
list(SORT SB_TEMPLATE_FILES)
set(SB_TEMPLATE_C "typedef struct { const char *path; const char *text; } SBTemplate;\nstatic const SBTemplate sb_templates[] = {\n")
foreach(SB_TEMPLATE_FILE IN LISTS SB_TEMPLATE_FILES)
    file(RELATIVE_PATH SB_REL "${CMAKE_CURRENT_SOURCE_DIR}/templates/brain" "${SB_TEMPLATE_FILE}")
    file(READ "${SB_TEMPLATE_FILE}" SB_TEXT)
    string(REPLACE "\\" "\\\\" SB_TEXT "${SB_TEXT}")
    string(REPLACE "\"" "\\\"" SB_TEXT "${SB_TEXT}")
    string(REPLACE "\r" "" SB_TEXT "${SB_TEXT}")
    string(REPLACE "\n" "\\n\"\n\"" SB_TEXT "${SB_TEXT}")
    string(APPEND SB_TEMPLATE_C "{\"${SB_REL}\", \"${SB_TEXT}\"},\n")
endforeach()
string(APPEND SB_TEMPLATE_C "};\n#define SB_TEMPLATE_COUNT (sizeof(sb_templates)/sizeof(sb_templates[0]))\n")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/sb_templates.h" "${SB_TEMPLATE_C}")

add_library(sb_core STATIC src/sb.c src/platform.c src/settings.c)
target_include_directories(sb_core PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
    target_compile_options(sb_core PRIVATE /W4 /utf-8)
    target_compile_definitions(sb_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(sb_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(sb_app_model STATIC app/model.c app/graph.c)
target_include_directories(sb_app_model PUBLIC app)
target_link_libraries(sb_app_model PUBLIC sb_core)
if(UNIX)
    target_link_libraries(sb_app_model PUBLIC m)
endif()
if(MSVC)
    target_compile_options(sb_app_model PRIVATE /W4 /utf-8)
    target_compile_definitions(sb_app_model PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(sb_app_model PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(SB_BUILD_UI)
    find_package(SDL3 3.2 CONFIG QUIET COMPONENTS SDL3-static)
    if(NOT SDL3_FOUND)
        include(FetchContent)
        if(POLICY CMP0135)
            cmake_policy(SET CMP0135 NEW)
        endif()
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TESTS OFF CACHE BOOL "" FORCE)
        set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
        set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
        set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
        set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
        set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
        set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(SDL3
            URL https://github.com/libsdl-org/SDL/releases/download/release-3.2.30/SDL3-3.2.30.tar.gz
            URL_HASH SHA256=4c3b09330d866dc52eb65b66259a6684ad387252ca8c57901b3a2b534eb42e3d)
        FetchContent_MakeAvailable(SDL3)
    endif()
    add_library(sb_ui STATIC app/ui.c app/space.c app/icons.c)
    target_include_directories(sb_ui PUBLIC app third_party/ui)
    target_link_libraries(sb_ui PUBLIC sb_core SDL3::SDL3-static)
    if(MSVC)
        target_compile_options(sb_ui PRIVATE /utf-8)
        target_compile_definitions(sb_ui PRIVATE _CRT_SECURE_NO_WARNINGS)
    elseif(UNIX)
        target_link_libraries(sb_ui PUBLIC m)
    endif()
    add_library(sb_desktop STATIC app/desktop.c app/dialog.c app/self_test.c app/keyboard_test.c)
    target_link_libraries(sb_desktop PUBLIC sb_ui sb_app_model)
    if(MSVC)
        target_compile_options(sb_desktop PRIVATE /W4 /utf-8)
        target_compile_definitions(sb_desktop PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(sb_desktop PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(WIN32)
        add_executable(secondbrain WIN32 app/main.c)
    elseif(APPLE)
        add_executable(secondbrain MACOSX_BUNDLE app/main.c)
        set_target_properties(secondbrain PROPERTIES
            MACOSX_BUNDLE_GUI_IDENTIFIER "org.secondbrain.desktop"
            MACOSX_BUNDLE_BUNDLE_NAME "SecondBrain"
            MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
            MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
    else()
        add_executable(secondbrain app/main.c)
    endif()
    target_link_libraries(secondbrain PRIVATE sb_desktop)
    if(MSVC)
        target_compile_options(secondbrain PRIVATE /utf-8)
        target_compile_definitions(secondbrain PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    if(APPLE)
        set(SB_ASSET_DEST "$<TARGET_BUNDLE_DIR:secondbrain>/Contents/Resources/assets")
    else()
        set(SB_ASSET_DEST "$<TARGET_FILE_DIR:secondbrain>/assets")
    endif()
    add_custom_command(TARGET secondbrain POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${SB_ASSET_DEST}"
        VERBATIM)
    install(TARGETS secondbrain BUNDLE DESTINATION . COMPONENT Runtime
        RUNTIME DESTINATION . COMPONENT Runtime)
    if(NOT APPLE)
        install(DIRECTORY assets DESTINATION . COMPONENT Runtime)
    endif()
    install(FILES third_party/licenses/SDL3.txt DESTINATION licenses COMPONENT Runtime)
    install(FILES third_party/ui/Nuklear-LICENSE DESTINATION licenses RENAME Nuklear.txt COMPONENT Runtime)
    install(FILES assets/fonts/OFL.txt DESTINATION licenses RENAME Noto.txt COMPONENT Runtime)
    install(FILES docs/QUICKSTART.txt LICENSE DESTINATION . COMPONENT Runtime)
    if(BUILD_TESTING)
        add_executable(secondbrain-preferences-tests tests/test_preferences.c)
        target_link_libraries(secondbrain-preferences-tests PRIVATE sb_desktop)
        if(MSVC)
            target_compile_options(secondbrain-preferences-tests PRIVATE /utf-8)
        endif()
        add_test(NAME preferences-restart COMMAND secondbrain-preferences-tests
            "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/NotoSans-Regular.ttf"
            "${CMAKE_CURRENT_BINARY_DIR}/preferences-data")
        set_tests_properties(preferences-restart PROPERTIES TIMEOUT 120)
        add_test(NAME preferences-process-restart COMMAND "${CMAKE_COMMAND}"
            "-DSB_EXECUTABLE=$<TARGET_FILE:secondbrain>"
            "-DSB_TEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/process-preferences"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_preferences_process.cmake")
        set_tests_properties(preferences-process-restart PROPERTIES TIMEOUT 90)
        add_executable(secondbrain-ui-tests tests/test_ui.c)
        target_link_libraries(secondbrain-ui-tests PRIVATE sb_ui)
        if(MSVC)
            target_compile_options(secondbrain-ui-tests PRIVATE /utf-8)
        endif()
        add_test(NAME ui-rendering-editor COMMAND secondbrain-ui-tests
            "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/NotoSans-Regular.ttf"
            "${CMAKE_CURRENT_BINARY_DIR}/ui-test.bmp")
        add_test(NAME desktop-workflow COMMAND secondbrain --self-test "${CMAKE_CURRENT_BINARY_DIR}/desktop-data")
        add_test(NAME keyboard-workflow COMMAND secondbrain --keyboard-test "${CMAKE_CURRENT_BINARY_DIR}/keyboard-data")
        set_tests_properties(keyboard-workflow PROPERTIES TIMEOUT 300)
        add_test(NAME project-example COMMAND secondbrain --workspace "${CMAKE_CURRENT_SOURCE_DIR}/brains"
            --project secondbrain --snapshot "${CMAKE_CURRENT_BINARY_DIR}/own-project.bmp")
        set_tests_properties(desktop-workflow project-example PROPERTIES TIMEOUT 240)
    endif()
endif()

add_executable(secondbrain-cli src/cli.c)
target_link_libraries(secondbrain-cli PRIVATE sb_core)

    install(TARGETS secondbrain-cli RUNTIME DESTINATION . COMPONENT Runtime)
