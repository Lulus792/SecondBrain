cmake_minimum_required(VERSION 3.20)
project(SecondBrain VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SB_BUILD_UI "Build SDL desktop UI" OFF)
include(CTest)

file(GLOB_RECURSE SB_TEMPLATE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/templates/brain/*.md")
list(SORT SB_TEMPLATE_FILES)
set(SB_TEMPLATE_C "typedef struct { const char *path; const char *text; } SBTemplate;\nstatic const SBTemplate sb_templates[] = {\n")
foreach(SB_TEMPLATE_FILE IN LISTS SB_TEMPLATE_FILES)
    file(RELATIVE_PATH SB_REL "${CMAKE_CURRENT_SOURCE_DIR}/templates/brain" "${SB_TEMPLATE_FILE}")
    file(READ "${SB_TEMPLATE_FILE}" SB_TEXT)
    string(REPLACE "\\" "\\\\" SB_TEXT "${SB_TEXT}")
    string(REPLACE "\"" "\\\"" SB_TEXT "${SB_TEXT}")
    string(REPLACE "\r" "" SB_TEXT "${SB_TEXT}")
    string(REPLACE "\n" "\\n\"\n\"" SB_TEXT "${SB_TEXT}")
    string(APPEND SB_TEMPLATE_C "{\"${SB_REL}\", \"${SB_TEXT}\"},\n")
endforeach()
string(APPEND SB_TEMPLATE_C "};\n#define SB_TEMPLATE_COUNT (sizeof(sb_templates)/sizeof(sb_templates[0]))\n")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/sb_templates.h" "${SB_TEMPLATE_C}")

add_library(sb_core STATIC src/sb.c src/platform.c)
target_include_directories(sb_core PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
    target_compile_options(sb_core PRIVATE /W4 /utf-8)
    target_compile_definitions(sb_core PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(sb_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(SB_BUILD_UI)
    find_package(SDL3 3.2 CONFIG QUIET)
    if(NOT SDL3_FOUND)
        include(FetchContent)
        if(POLICY CMP0135)
            cmake_policy(SET CMP0135 NEW)
        endif()
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TESTS OFF CACHE BOOL "" FORCE)
        set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
        set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
        set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
        set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
        set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(SDL3
            URL https://github.com/libsdl-org/SDL/releases/download/release-3.2.30/SDL3-3.2.30.tar.gz
            URL_HASH SHA256=4c3b09330d866dc52eb65b66259a6684ad387252ca8c57901b3a2b534eb42e3d)
        FetchContent_MakeAvailable(SDL3)
    endif()
    add_library(sb_ui STATIC app/ui.c)
    target_include_directories(sb_ui PUBLIC app third_party/ui)
    target_link_libraries(sb_ui PUBLIC sb_core SDL3::SDL3)
    if(MSVC)
        target_compile_options(sb_ui PRIVATE /utf-8)
        target_compile_definitions(sb_ui PRIVATE _CRT_SECURE_NO_WARNINGS)
    elseif(UNIX)
        target_link_libraries(sb_ui PUBLIC m)
    endif()
    if(BUILD_TESTING)
        add_executable(secondbrain-ui-tests tests/test_ui.c)
        target_link_libraries(secondbrain-ui-tests PRIVATE sb_ui)
        if(MSVC)
            target_compile_options(secondbrain-ui-tests PRIVATE /utf-8)
        endif()
        add_test(NAME ui-rendering-editor COMMAND secondbrain-ui-tests
            "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/NotoSans-Regular.ttf"
            "${CMAKE_CURRENT_BINARY_DIR}/ui-test.bmp")
    endif()
endif()

add_executable(secondbrain-cli src/cli.c)
target_link_libraries(secondbrain-cli PRIVATE sb_core)
if(MSVC)
    target_compile_options(secondbrain-cli PRIVATE /utf-8)
elseif(MINGW)
    target_link_options(secondbrain-cli PRIVATE -municode)
endif()

if(BUILD_TESTING)
    add_executable(secondbrain-core-tests tests/test_core.c)
    target_link_libraries(secondbrain-core-tests PRIVATE sb_core)
    if(MSVC)
        target_compile_options(secondbrain-core-tests PRIVATE /utf-8)
    endif()
    add_test(NAME core-workflow COMMAND secondbrain-core-tests "${CMAKE_CURRENT_BINARY_DIR}/test-data")
endif()
